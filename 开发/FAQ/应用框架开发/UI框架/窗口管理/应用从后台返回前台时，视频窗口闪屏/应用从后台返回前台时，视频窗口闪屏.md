# 应用从后台返回前台时，视频窗口闪屏

更新时间：2026-06-26 07:47:42

来源：https://developer.huawei.com/consumer/cn/doc/harmonyos-faqs/faqs-arkui-862

#### 问题现象

应用从后台返回前台时，视频窗口内容闪过其他内容。
 

![](https://contentcenter-vali-drcn.dbankcdn.cn/pvt_2/DeveloperAlliance_scene_100_1/b0/v3/izFZzevxR2K1uMCFDTHgIw/zh-cn_image_0000002628398894.png?HW-CC-KV=V1&HW-CC-Date=20260730T072420Z&HW-CC-Expire=86400&HW-CC-Sign=26427A74B72BC84606C2C8A0CE3A5D3081A3D31767899D67BEC61B2F063C9D79)

 
 

#### 背景知识

- [onPageShow](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-custom-component-lifecycle#onpageshow)/[onPageHide](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-custom-component-lifecycle#onpagehide)在页面跳转、应用进入/退出前台等场景时触发一次。
- [Video](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-media-components-video)为用于播放视频文件并控制其播放状态的组件。

 
 

#### 问题定位
1. 使用DevEco Testing查看问题组件，为Stack组件下的Video组件，且同层组件为Image组件。
![](https://contentcenter-vali-drcn.dbankcdn.cn/pvt_2/DeveloperAlliance_scene_100_1/77/v3/0n3Kk9WLSGWaM7tkCAbnlA/zh-cn_image_0000002658798169.png?HW-CC-KV=V1&HW-CC-Date=20260730T072420Z&HW-CC-Expire=86400&HW-CC-Sign=EA0871229836A0BDC696F62B180AF1329E021EE524CC12768D6697C74BC0F152)

2. 查看该页面的相关设置，发现页面使用了onPageShow和onPageHide在应用进入和退出前台时控制图片的显示。应用退出前台暂停播放时显示静态图片，返回前台恢复播放时又从图片切换到视频，造成视频窗口闪屏。
```text
import { window } from '@kit.ArkUI';
import { common } from '@kit.AbilityKit';

@Entry
@Component
export struct Index {
  private videoSrc: ResourceStr = $rawfile('video1.mp4'); <em>// $rawfile('video1.mp4')需要替换为开发者需要的视频资源文件</em>
  controller: VideoController = new VideoController();
  @State showImg: boolean = true;
  context: common.UIAbilityContext = this.getUIContext().getHostContext() as common.UIAbilityContext;

  aboutToAppear(): void {
    window.getLastWindow(this.context).then((lastWindow) => {
      lastWindow.setWindowLayoutFullScreen(true);
      let systemBarProperties: window.SystemBarProperties = {
        statusBarColor: '#00000000',
        statusBarContentColor: '#ffffff'
      };
      lastWindow.setWindowSystemBarProperties(systemBarProperties);
    });
  }

  onPageHide(): void {
  <em>  // 应用退到后台时显示预览图和暂停视频</em>
    this.showImg = true;
    this.controller.pause();
  }

  onPageShow(): void {
    <em>// 应用返回时播放视频</em>
<em>    // 模拟视频加载</em>
    setTimeout(() => {
      this.controller.start();
    }, 300);
  }

  build() {
    Stack({ alignContent: Alignment.Start }) {
      Video({
        src: this.videoSrc,
        controller: this.controller,
      })
        .width('100%')
        .height('90%')
        .objectFit(ImageFit.Contain)
        .loop(true)
        .autoPlay(true)
        .onStart(() => {
          this.showImg = false; <em>// 开始播放视频时不显示预览图</em>
        });
      if (this.showImg) {
        Image($r('app.media.myposter1')) <em>// $r('app.media.myposter1')需要替换为开发者需要的图片资源文件</em>
          .width('100%')
          .height('90%')
          .objectFit(ImageFit.Contain);
      }
    }
    .height('100%')
    .width('100%')
    .backgroundColor(Color.Black)
    .expandSafeArea([SafeAreaType.SYSTEM], [SafeAreaEdge.TOP, SafeAreaEdge.BOTTOM]); <em>// 扩展页面区域到导航栏和状态栏</em>
  }
}
```

 
 

#### 分析结论

应用退出前台暂停播放时显示静态图片，返回前台恢复播放时又从图片切换到视频，造成视频窗口闪屏。
 
 

#### 修改建议

应用进入和退出前台时不改变图片的显示，只在初次播放时改变。
 
```text
import { window } from '@kit.ArkUI';
import { common } from '@kit.AbilityKit';

interface DurationObject {
  duration: number;
}

interface TimeObject {
  time: number;
}

@Entry
@Component
export struct Index {
  private videoSrc: ResourceStr = $rawfile('video1.mp4'); <em>// $rawfile('video1.mp4')需要替换为开发者需要的视频资源文件</em>
  controller: VideoController = new VideoController();
  @State showImg: boolean = true;
  @State isPlay: boolean = false;
  @State durationTime: number = 0;
  @State currentTime: number = 0;
  @State sliderWidth: number = 230;
  @State @Watch('changeOrientation') isFull: boolean = false;
  context: common.UIAbilityContext = this.getUIContext().getHostContext() as common.UIAbilityContext;

  aboutToAppear(): void {
    window.getLastWindow(this.context).then((lastWindow) => {
      lastWindow.setWindowLayoutFullScreen(true);
      let systemBarProperties: window.SystemBarProperties = {
        statusBarColor: '#00000000',
        statusBarContentColor: '#ffffff'
      };
      lastWindow.setWindowSystemBarProperties(systemBarProperties);
    });
  }

  timeConvert(time: number): string {
    let min: number = Math.floor(time / 60);
    let second: string = (time % 60).toFixed(0);
    second = second.padStart(2, '0');
    return `${min}:${second}`;
  }

  changeOrientation() {
    let context: common.UIAbilityContext = this.getUIContext().getHostContext() as common.UIAbilityContext;
    window.getLastWindow(context).then((lastWindow) => {
      lastWindow.setPreferredOrientation(this.isFull ? window.Orientation.AUTO_ROTATION_LANDSCAPE :
        window.Orientation.PORTRAIT);
     <em> // 进入全屏时进入沉浸式，退出全屏时再显示状态栏和导航栏</em>
      lastWindow.setWindowLayoutFullScreen(this.isFull);
      lastWindow.setWindowSystemBarEnable(this.isFull ? [] : ['status', 'navigation']);
    });
  }

  onPageHide(): void {
   <em> // 应用退到后台时暂停视频</em>
    this.controller.pause();
  }

  onPageShow(): void {
 <em>   // 应用返回时播放视频</em>
<em>    // 模拟视频加载</em>
    setTimeout(() => {
      this.controller.start();
    }, 300);
  }

  build() {
    Stack({ alignContent: Alignment.Start }) {
      Video({
        src: this.videoSrc,
        controller: this.controller,
      })
        .width('100%')
        .height('100%')
        .objectFit(ImageFit.Contain)
        .controls(false)
        .loop(true)
        .autoPlay(true)
        .onPrepared((e?: DurationObject) => {
          this.showImg = false;<em> // 加载完成视频时不显示预览图</em>
          console.info('duration is ' + e?.duration);
          this.isPlay = true;
          if (e !== undefined) {
            this.durationTime = e.duration;
          }
        })
        .onUpdate((e?: TimeObject) => {
          if (e !== undefined) {
            this.currentTime = e.time;
          }
        })
        .onSizeChange(() => {
          this.sliderWidth = this.isFull ? 650 : 230;
        });

      Stack() {
        Row() {
          <em>// $r('app.media.ic_video_play')和$r('app.media.ic_video_pause')需要替换为开发者需要的图片资源文件</em>
          Image(this.isPlay ? $r('app.media.ic_video_play') : $r('app.media.ic_video_pause'))
            .width(25)
            .height(25)
            .onClick(() => {
              if (this.isPlay) {
                this.controller.pause();
                this.isPlay = false;
              } else {
                this.controller.start();
                this.isPlay = true;
              }
            });

          <em>// 左侧时间</em>
          Text(this.timeConvert(this.currentTime))
            .fontColor(Color.White)
            .textAlign(TextAlign.End)
            .fontWeight(FontWeight.Regular)
            .margin({ left: 10 });

          Slider({
            value: this.currentTime,
            min: 0,
            max: this.durationTime,
            style: SliderStyle.OutSet
          })
            .blockColor(Color.White)
            .trackColor(Color.Gray)
            .selectedColor('#007DFF')
            .showTips(false)
            .width(this.sliderWidth)
            .onChange((value: number, mode: SliderChangeMode) => {
              if (mode === SliderChangeMode.Begin) {
                this.controller.pause();
              } else if (mode === SliderChangeMode.Moving) {
                this.controller.setCurrentTime(value);
              } else if (mode === SliderChangeMode.End) {
                this.controller.start();
              }
            });

         <em> // 右侧时间</em>
          Text(this.timeConvert(this.durationTime))
            .id('durationTimeText')
            .fontColor(Color.White)
            .fontWeight(FontWeight.Regular);

         <em> // $r('app.media.out')和$r('app.media.full')需要替换为开发者需要的图片资源文件</em>
          Image(this.isFull ? $r('app.media.out') : $r('app.media.full'))
            .width(20)
            .height(20)
            .objectFit(ImageFit.Contain)
            .onClick(() => {
              this.isFull = !this.isFull; <em>// 进入全屏或退出全屏</em>
            })
            .margin({ left: 5 });
        }
        .width('100%')
        .justifyContent(FlexAlign.Center)
        .padding({ left: 10, right: 5 })
        .margin({ bottom: 5 });
      }
      .width('100%')
      .height('100%')
      .alignContent(Alignment.Bottom);

      if (this.showImg) {
        Image($r('app.media.myposter1')) <em>// $r('app.media.myposter1')需要替换为开发者需要的图片资源文件</em>
          .width('100%')
          .height('90%')
          .objectFit(ImageFit.Contain);
      }
    }
    .height('100%')
    .width('100%')
    .backgroundColor(Color.Black)
    .expandSafeArea([SafeAreaType.SYSTEM], [SafeAreaEdge.TOP, SafeAreaEdge.BOTTOM]); <em>// 扩展页面区域到导航栏和状态栏</em>
  }
}
```
