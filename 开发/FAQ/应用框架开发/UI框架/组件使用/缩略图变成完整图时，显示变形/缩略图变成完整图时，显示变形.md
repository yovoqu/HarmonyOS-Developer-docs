# 缩略图变成完整图时，显示变形

更新时间：2026-06-26 07:47:42

来源：https://developer.huawei.com/consumer/cn/doc/harmonyos-faqs/faqs-arkui-835

#### 问题现象

点击缩略图显示完整图片时，图片显示变形。
 

![](https://contentcenter-vali-drcn.dbankcdn.cn/pvt_2/DeveloperAlliance_scene_100_1/33/v3/WBKRkmESTEyqIT8TLNQM4w/zh-cn_image_0000002628398446.png?HW-CC-KV=V1&HW-CC-Date=20260811T005830Z&HW-CC-Expire=86400&HW-CC-Sign=621519FFD110ECB867BFCAA5378A90F1A6EA3C6E3611FBFD58DC533C6C3D01E2)

 
 

#### 背景知识

- [Image](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-image)为图片组件，常用于在应用中显示图片。
- 设置Image组件的[objectFit](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-image#objectfit)属性可调整图片的填充效果。
- 通过Image组件的[onComplete](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-image#oncomplete)事件，可在图片数据加载成功和解码成功时获取图片尺寸。

 
 

#### 问题定位
1. 使用DevEco Testing查看问题组件，问题组件为Image组件。
![](https://contentcenter-vali-drcn.dbankcdn.cn/pvt_2/DeveloperAlliance_scene_100_1/97/v3/Z3EbjZEiSAmZmcGR1MKh9Q/zh-cn_image_0000002658797725.png?HW-CC-KV=V1&HW-CC-Date=20260811T005830Z&HW-CC-Expire=86400&HW-CC-Sign=9FD9F09AD254CCAD94B96ACA8FA16E30CC81291FC8EFDB449E919C1B517047B2)

2. 查看该Image组件的设置，显示完整图片时的组件尺寸设置不合理，高度为屏幕的高度而不是等比例缩放下完整图片的高度。
```text
<em>// 全屏显示图片</em>
@Builder
export function DialogBuilder() {
  Dialog();
}

@Component
export struct Dialog {
  pageInfos: NavPathStack = new NavPathStack();

  build() {
    NavDestination() {
      Stack() {
        Image($r('app.media.imggroup2')) <em>// $r('app.media.imggroup2')需要更换为开发者需要的图片资源</em>
          .width('80%')
          .height('100%') <em>// 高度与屏幕一致</em>
          .objectFit(ImageFit.Fill); <em>// 图片充满显示边界</em>
      }
      .height('100%')
    }
    .height('100%')
  }
}
```

 
 

#### 分析结论

显示完整图片时的Image组件尺寸设置不合理，导致图片显示变形。
 
 

#### 修改建议

不设置Image组件的高度，按照设置的宽度进行等比例缩放。
 
```text
import { window } from '@kit.ArkUI';
import { common } from '@kit.AbilityKit';

@Entry
@Component
struct NavigationCustomTransitionExample {
  pageInfos: NavPathStack = new NavPathStack();

  aboutToAppear() {
    this.pageInfos.pushPath({ name: 'PageMain' }, false);
  }

  build() {
    Stack() {
      Navigation(this.pageInfos)
        .hideNavBar(true)
        .width('100%')
        .height('100%');
    }
    .width('100%')
    .height('100%')
    .expandSafeArea([SafeAreaType.SYSTEM], [SafeAreaEdge.TOP, SafeAreaEdge.BOTTOM]);
  }
}

@Builder
export function PageMainBuilder() {
  PageMain();
}

<em>// 全屏显示图片</em>
@Builder
export function DialogBuilder() {
  Dialog();
}

@Component
export struct Dialog {
  pageInfos: NavPathStack = new NavPathStack();

  aboutToAppear(): void {
    let context: common.UIAbilityContext = this.getUIContext().getHostContext() as common.UIAbilityContext;
    window.getLastWindow(context).then((lastWindow) => {
      lastWindow.setWindowSystemBarEnable([]);
    });
  }

  aboutToDisappear(): void {
    let context: common.UIAbilityContext = this.getUIContext().getHostContext() as common.UIAbilityContext;
    window.getLastWindow(context).then((lastWindow) => {
      lastWindow.setWindowSystemBarEnable(['status', 'navigation']);
    });
  }

  build() {
    NavDestination() {
      Stack() {
       <em> // 图片过长时可滑动</em>
        Scroll() {
          Image($r('app.media.imggroup2')) <em>// $r('app.media.imggroup2')需要更换为开发者需要的图片资源</em>
            .objectFit(ImageFit.Contain)
            .width('80%');
         <em> // 不设置Image组件的高度，按照设置的宽度进行等比例缩放</em>
        }
        .height('100%')
        .width('100%');
      }
      .height('100%')
      .width('100%')
      .onClick(() => {
        this.pageInfos.pop(); <em>// 完整图时点击缩小，显示缩略图</em>
      })
      .backgroundColor(Color.Black)
      .expandSafeArea([SafeAreaType.SYSTEM], [SafeAreaEdge.TOP, SafeAreaEdge.BOTTOM]);
    }
    .hideTitleBar(true)
    .onReady((context: NavDestinationContext) => {
      this.pageInfos = context.pathStack;
    })
    .height('100%')
    .width('100%')
    .mode(NavDestinationMode.DIALOG)
    .systemTransition(NavigationSystemTransitionType.NONE)
    .expandSafeArea([SafeAreaType.SYSTEM], [SafeAreaEdge.TOP, SafeAreaEdge.BOTTOM]);
  }
}

@Component
export struct PageMain {
  pageInfos: NavPathStack = new NavPathStack();

  build() {
    NavDestination() {
      Stack() {
        Image($r('app.media.imggroup2')) <em>// $r('app.media.imggroup2')需要更换为开发者需要的图片资源</em>
          .width(300)
          .height(300)
          .objectFit(ImageFit.CENTER)
          .onClick(() => {
            this.pageInfos.pushPath({ name: 'Dialog' }); <em>// 缩略图时点击放大，显示完整图片</em>
          });
      }
      .width('100%')
      .height('100%');
    }
    .hideTitleBar(true)
    .onReady((context: NavDestinationContext) => {
      this.pageInfos = context.pathStack;
    })
    .height('100%')
    .width('100%')
    .mode(NavDestinationMode.STANDARD);
  }
}
```
