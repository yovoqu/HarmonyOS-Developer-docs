# 如何实现想法标记功能

更新时间：2026-06-26 09:07:13

来源：https://developer.huawei.com/consumer/cn/doc/harmonyos-faqs/faqs-arkui-1027

#### 问题现象

想要实现类似于华为阅读应用的想法标记功能，具体要求如下：
 1. 长按选择文本，弹出的菜单内包含想法按钮。
2. 点击想法按钮弹出想法输入框，当输入框为空时不能添加想法。
3. 点击添加想法后，输入框消失，选择的文本标记高亮。
4. 点击高亮部分可以显示对应的想法内容。

#### 背景知识

[Text](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-text)是文本组件，用于展示用户视图，如显示文章的文字内容。该组件支持绑定自定义文本选择菜单，用户可根据需要选择不同功能。此外，还可以扩展自定义菜单，丰富可用选项，进一步提升用户体验。[Span](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-span)则用于展示行内文本。
 
 

#### 解决方案

由于[RichEditor](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-richeditor)组件不支持禁止文本编辑，所以主体的文本展示功能使用Text和Span组件实现。按照要求，可以将想法标记功能分为以下几点分别实现：
 1. 选择菜单内添加自定义按钮：通过设置Text组件的[editMenuOptions](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-text#editmenuoptions12)属性实现。
2. 点击想法按钮弹出想法输入框：[绑定半模态页面](https://developer.huawei.com/consumer/cn/doc/harmonyos-guides/arkts-sheet-page)，将输入框放在半模态页面内，点击按钮时打开。
3. 输入框为空时不能添加想法：通过[TextInput](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-textinput)组件的[onChange()](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/ts-basic-components-textinput#onchange)事件读取输入框的内容并记录，在添加想法时判断输入框内容是否为空，若为空则不做任何反应。
4. 将选择的文本标记高亮：参考[文本标记高亮显示](https://developer.huawei.com/consumer/cn/doc/architecture-guides/text_marker_ability-0000002283796046)。
5. 点击高亮部分显示对应想法：创建一个Map用来存储想法，以高亮的结束坐标为键，想法内容为值，点击高亮时读取Map值用于显示。
 
完整示例代码如下：
 
```json
import { PromptAction } from '@kit.ArkUI';
import { JSON } from '@kit.ArkTS';

const TEXT_DATA = '文本内容'; <em>// </em><em>展示的文本内容</em>

@Entry
@Component
struct TextPage {
  promptAction: PromptAction = this.getUIContext().getPromptAction();
  @StorageProp('bottomRectHeight')
  bottomRectHeight: number = 0;
  @StorageProp('topRectHeight')
  topRectHeight: number = 0;
  @State markList: HighlightMark[] = [];<em> </em><em>// 高亮标记数据</em>
  currentColor: number = 0;<em> </em><em>// 当前笔记颜色</em>
  @State textLineHeight: Length = 26;<em> </em><em>// 行高</em>
  @State textFontSize: Length = 17; <em>// </em><em>字体大小</em>
  originalText: string = TEXT_DATA; <em>// </em><em>显示文本</em>
  private selectionStart: number = -1; <em>// </em><em>当前选中区间开头</em>
  private selectionEnd: number = -1; <em>// 当前选中区间结尾</em>
  private controller: TextController = new TextController();
  @State isShowSheet: boolean = false;<em> </em><em>// 控制想法输入框的显隐</em>
  @State ideaInput: string = '';<em> </em><em>// 记录输入的想法</em>
  onCreateMenu = (menuItems: Array<TextMenuItem>) => { <em>// </em><em>自定义菜单内容</em>
    let item1: TextMenuItem = {
      content: '想法',
      icon: $r('app.media.startIcon'),
      id: TextMenuItemId.of('IDEA'),
    };
    menuItems.unshift(item1);
    return menuItems;
  };
  onMenuItemClick = (menuItem: TextMenuItem) => { <em>// </em><em>自定义点击菜单事件</em>
    if (menuItem.id.equals(TextMenuItemId.of('IDEA'))) {
      this.isShowSheet = true; <em>// 打开想法输入框</em>
      return true;
    }
    return false;
  };
  @State editMenuOptions: EditMenuOptions = {
    onCreateMenu: this.onCreateMenu,
    onMenuItemClick: this.onMenuItemClick
  };
  ideaSet: Map<number, string> = new Map(); <em>// </em><em>创建一个集合来保存所有想法</em>

 <em> // 初始化文本内容</em>
  aboutToAppear(): void {
    this.markList = [{
     <em> // 载入文本内容</em>
      start: 0,
      end: this.originalText.length,
      color: Color.Transparent
    }];
  }

  <em>// 标记选中区域高亮</em>
  markHighlight(highlight: boolean) {
    if (this.selectionStart !== -1 && this.selectionEnd !== -1) {
      let leftMarks: HighlightMark[] = [];<em> </em><em>// 区域左侧的文本样式数据</em>
      let rightMarks: HighlightMark[] = []; <em>// 区域右侧的文本样式数据</em>
      <em>// 当前需要更改样式的区域</em>
      let currentSelection: HighlightMark = {
        start: this.selectionStart,
        end: this.selectionEnd,
        color: highlight ? Color.Pink : Color.Transparent
      };

      this.markList.forEach((mark) => {
       <em> // 如果既有样式区间开头在当前选中区域开头之前</em>
        if (mark.start < currentSelection.start) {
         <em> // 如果既有样式区间结尾在当前选中区域开头之前，即两个区间无交集</em>
          if (mark.end < currentSelection.start) {
            leftMarks.push({
              start: mark.start,
              end: mark.end,
              color: mark.color
            });
          } else if (mark.color === currentSelection.color) {
          <em>  // 两个区间有交集且样式一致，合并入当前选中区域</em>
            currentSelection.start = mark.start;
          } else {
           <em> // 样式不一致，截断区间</em>
            leftMarks.push({
              start: mark.start,
              end: currentSelection.start,
              color: mark.color
            });
          }
        }
       <em> // 既有样式区间在当前选中区域末尾之后</em>
        if (mark.end > currentSelection.end) {
        <em>  // 如果既有样式区间开头在当前选中区域开头之后，即两个区间无交集</em>
          if (mark.start > currentSelection.end) {
            rightMarks.push({
              start: mark.start,
              end: mark.end,
              color: mark.color
            });
          } else if (mark.color === currentSelection.color) {
           <em> // 两个区间有交集且样式一致，合并入当前选中区域</em>
            currentSelection.end = mark.end;
          } else {
         <em>   // 样式不一致，截断区间</em>
            rightMarks.push({
              start: currentSelection.end,
              end: mark.end,
              color: mark.color
            });
          }
        }
      });
    <em>  // 合并当前与左右文本样式数据</em>
      this.markList = leftMarks.concat(currentSelection, rightMarks);
    }
  }

  <em>// 文字预览区域</em>
  @Builder
  TextViewer() {
    Scroll() {
      Text('', { controller: this.controller }) {
        ForEach(this.markList, (mark: HighlightMark) => {
          Span(this.originalText.substring(mark.start, mark.end))
            .textBackgroundStyle({ color: mark.color, radius: 0 })
            .lineHeight(this.textLineHeight)
            .fontSize(this.textFontSize)
            .onClick(() => { <em>// </em><em>通过底色判断是否绑定想法</em>
              if (mark.color != Color.Transparent) {
                this.promptAction.showToast({ message: this.ideaSet.get(mark.end) }); <em>// </em><em>展示想法内容</em>
              }
            })
        }, (mark: HighlightMark) => JSON.stringify(mark));
      }
      .editMenuOptions(this.editMenuOptions)<em> </em><em>// 绑定自定义菜单</em>
      .padding(24)
      .copyOption(CopyOptions.InApp)
      .onTextSelectionChange((selectionStart: number, selectionEnd: number) => { <em>// 记录光标选中区域</em>
        this.selectionStart = selectionStart;
        this.selectionEnd = selectionEnd;
      });

    }
    .scrollBar(BarState.Off)
    .width('90%')
    .layoutWeight(1);
  }

 <em> // 想法标题</em>
  @Builder
  titleBuilder() {
    Column() {
      Text('请写下你的想法')
    }
    .width('100%')
    .justifyContent(FlexAlign.Center)
  }

 <em> // 想法输入框</em>
  @Builder
  SheetBuilder() {
    Column({ space: 30 }) {
      TextInput()
        .width('90%')
        .height('70%')
        .onChange((index: string) => {
          this.ideaInput = index;<em> </em><em>// 记录输入内容</em>
        })

      Button('添加想法')
        .backgroundColor('#0A59F7')
        .opacity(this.ideaInput ? 1 : 0.4)
        .width('90%')
        .onClick(() => {
          if (this.ideaInput) { <em>// 设置有内容时才可以点击</em>
            this.markHighlight(true); <em>// 标记高亮表示此处有想法</em>
            this.ideaSet.set(this.selectionEnd, this.ideaInput);<em> </em><em>// 将想法放入集合</em>
            this.ideaInput = '';<em> </em><em>// 清空输入记录</em>
            this.isShowSheet = false; <em>// </em><em>关闭想法输入框</em>
            this.controller.closeSelectionMenu(); <em>// </em><em>关闭菜单</em>
          }
        })
    }
    .width('100%')
    .height('100%')
  }

  build() {
    Column() {
      this.TextViewer();
    }
    .height('100%')
    .width('100%')
    .padding({
      top: this.getUIContext().px2vp(this.topRectHeight),
      bottom: this.getUIContext().px2vp(this.bottomRectHeight)
    })
    .bindSheet($$this.isShowSheet, this.SheetBuilder(), {
      detents: [SheetSize.MEDIUM, SheetSize.LARGE, 600],
      preferType: SheetType.BOTTOM,
      title: this.titleBuilder(),
      showClose: false,
      dragBar: false
    })
  }
}

interface HighlightMark {<em> </em><em>// 创建高亮文字类</em>
  start: number; <em>// 区间开头（含）</em>
  end: number; <em>// 区间结尾（不含）</em>
  color: ResourceColor; <em>// 高亮颜色，无高亮标记则为透明</em>
}
```
