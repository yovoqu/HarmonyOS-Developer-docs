# 如何适配网页内播放器全屏

更新时间：2026-06-26 09:07:13

来源：https://developer.huawei.com/consumer/cn/doc/harmonyos-faqs/faqs-arkweb-71

在工程中的module.json5配置文件中添加网络访问权限ohos.permission.INTERNET。
 
具体实现可参考如下代码
 
```text
import { mediaquery, window } from '@kit.ArkUI';
import { common } from '@kit.AbilityKit';
import { webview } from '@kit.ArkWeb';

@Entry
@Component
struct WebPlayerFullScreen {
  @State color: string = '#DB7093';
  @State text: string = 'Portrait';
  @State portraitFunc: mediaquery.MediaQueryResult | void | null = null;
 <em> // Full-screen exit processor, used to control the exit in full-screen state</em>
  handler: FullScreenExitHandler | null = null;
 <em> // The condition is met when the device is in landscape mode</em>
  listener: mediaquery.MediaQueryListener = this.getUIContext().getMediaQuery().matchMediaSync('(orientation: landscape)');
  controller: webview.WebviewController = new webview.WebviewController();

  onPortrait(mediaQueryResult: mediaquery.MediaQueryResult) {
   <em> // If the device is in landscape mode, change the corresponding page layout</em>
    if (mediaQueryResult.matches as boolean) {
      this.color = '#FFD700';
      this.text = 'Landscape';
    } else {
      this.color = '#DB7093';
      this.text = 'Portrait';
    }
  }

  aboutToAppear() {
  <em>  // Bind the current application instance</em>
<em>    // Bind callback function</em>
    this.listener.on('change', (mediaQueryResult: mediaquery.MediaQueryResult) => {
      this.onPortrait(mediaQueryResult);
    });
  }

 <em> // Change the horizontal and vertical screen status function of the device</em>
  private changeOrientation(isLandscape: boolean) {
  <em>  // Retrieve contextual information for UIAbility instances</em>
    let context: common.UIAbilityContext = this.getUIContext().getHostContext() as common.UIAbilityContext;
   <em> // Call this interface to manually change the device's horizontal and vertical screen status</em>
    window.getLastWindow(context).then((lastWindow) => {
      lastWindow.setPreferredOrientation(isLandscape ? window.Orientation.LANDSCAPE : window.Orientation.PORTRAIT);
    });
  }

  build() {
    Column() {
      Web({ src: 'https://developer.huawei.com/consumer/cn/design/', controller: this.controller })
        .javaScriptAccess(true)
        .domStorageAccess(true)
        .onFullScreenEnter((event) => {
          this.handler = event.handler;
          this.changeOrientation(true);
        })
        .onFullScreenExit(() => {
          if (this.handler) {
            this.handler.exitFullScreen();
            this.changeOrientation(false);
          }
        })
    }
    .width('100%')
    .height('100%')
  }
}
```
