# 如何获取系统版本号

更新时间：2026-06-26 07:48:29

来源：https://developer.huawei.com/consumer/cn/doc/harmonyos-faqs/faq-basics-service-kit-10

可通过[@ohos.deviceInfo (设备信息)](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/js-apis-device-info)模块，查询设备信息。
 
示例如下：
 
```text
import { deviceInfo } from '@kit.BasicServicesKit';
import { hilog } from '@kit.PerformanceAnalysisKit';

let displayVersionInfo: string = deviceInfo.displayVersion; <em>// product version</em>
let osFullNameInfo: string = deviceInfo.osFullName; <em>// system version</em>
let versionIdInfo: string = deviceInfo.versionId; <em>// version ID</em>

@Entry
@Component
struct GetSysVersion {
  build() {
    Row() {
      Column() {
        Button('get device info')
          .onClick(() => {
            hilog.info(0x000, 'displayVersion TAG', `displayVersion:${displayVersionInfo}`);
            hilog.info(0x000, 'osFullName TAG', `osFullName:${osFullNameInfo}`);
            hilog.info(0x000, 'versionId TAG', `versionId:${versionIdInfo}`);
          })
      }
      .width('100%')
    }
    .height('100%')
  }
}
```
